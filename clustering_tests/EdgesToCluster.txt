{"nodes":2,"edges":{"0":[{"to":1,"weight":1.5}],"1":[{"to":0,"weight":1.5}]},"totalWeight":1.5,"node":0,"cluster":[]}
@ cluster is empty → no neighbors counted → sum = 0

# Case 2: cluster contains only the node itself
{"nodes":1,"edges":{"0":[]},"totalWeight":0,"node":0,"cluster":[0]}
@ node is excluded from clusterSet, so no edges are counted → sum = 0
{"sum":0}

# Case 3: node connected to multiple nodes in cluster
{"nodes":3,"edges":{"0":[{"to":1,"weight":1.0},{"to":2,"weight":2.0}],"1":[{"to":0,"weight":1.0}],"2":[{"to":0,"weight":2.0}]},"totalWeight":3.0,"node":0,"cluster":[1,2]}
@ edges 0→1 (1.0) and 0→2 (2.0) both go to nodes in cluster → sum = 3.0
{"sum":3.0}

# Case 4: node connected to some nodes in cluster and some outside
{"nodes":4,"edges":{"0":[{"to":1,"weight":1.0},{"to":2,"weight":2.0},{"to":3,"weight":3.0}],"1":[{"to":0,"weight":1.0}],"2":[{"to":0,"weight":2.0}],"3":[{"to":0,"weight":3.0}]},"totalWeight":6.0,"node":0,"cluster":[2]}
@ only edge 0→2 (2.0) goes to a node in cluster → sum = 2.0
{"sum":2.0}

# Case 5: multiple edges from node to the same cluster member
{"nodes":2,"edges":{"0":[{"to":1,"weight":0.5},{"to":1,"weight":1.5}],"1":[{"to":0,"weight":2.0}]},"totalWeight":2.0,"node":0,"cluster":[1]}
