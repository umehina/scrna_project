# Case 1: both empty
{"a":[],"b":[]}
@ two empty slices are equal
{"equal":true}

# Case 2: different lengths
{"a":[0,1],"b":[0,1,2]}
@ different lengths → not equal
{"equal":false}

# Case 3: same length, same elements
{"a":[0,1,1,2],"b":[0,1,1,2]}
@ identical contents → equal
{"equal":true}

# Case 4: same length, different element at one position
{"a":[0,1,1,2],"b":[0,1,2,2]}
@ differ at index 2 → not equal
{"equal":false}

# Case 5: negative numbers and non-contiguous values
{"a":[-1,3,3,10],"b":[-1,3,3,10]}
@ identical contents with negative/large values → equal
{"equal":true}
