# Case 1: empty cluster
{"nodes":3,"edges":{"0":[{"to":1,"weight":1.0}],"1":[{"to":0,"weight":1.0},{"to":2,"weight":2.0}],"2":[{"to":1,"weight":2.0}]},"totalWeight":3.0,"cluster":[]}
@ no nodes in cluster → sum over nothing → degree = 0
{"degree":0}

# Case 2: single node with no edges
{"nodes":1,"edges":{"0":[]},"totalWeight":0,"cluster":[0]}
@ node 0 has no edges → degree = 0
{"degree":0}

# Case 3: single node with multiple outgoing edges
{"nodes":3,"edges":{"0":[{"to":1,"weight":1.0},{"to":2,"weight":2.5}],"1":[{"to":0,"weight":1.0}],"2":[{"to":0,"weight":2.5}]},"totalWeight":3.5,"cluster":[0]}
@ node 0 has edges of weights 1.0 and 2.5 → degree = 3.5
{"degree":3.5}

# Case 4: two nodes with symmetric edges between them
{"nodes":2,"edges":{"0":[{"to":1,"weight":1.0}],"1":[{"to":0,"weight":1.0}]},"totalWeight":1.0,"cluster":[0,1]}
@ edges 0→1 and 1→0 are both counted → degree = 1.0 + 1.0 = 2.0
{"degree":2.0}

# Case 5: self-loop counted once
{"nodes":3,"edges":{"0":[{"to":1,"weight":1.0}],"1":[{"to":0,"weight":1.0}],"2":[{"to":2,"weight":3.0}]},"totalWeight":4.0,"cluster":[2]}
@ node 2 has a self-loop of weight 3.0 → degree = 3.0
{"degree":3.0}
