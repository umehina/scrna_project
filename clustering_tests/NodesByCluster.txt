# Case 1: empty partition
{"partition":[]}
@ no nodes → empty grouping map
{"groups":{}}

# Case 2: two clusters, balanced
{"partition":[0,0,1,1]}
@ nodes 0,1 in cluster 0 and 2,3 in cluster 1
{"groups":{"0":[0,1],"1":[2,3]}}

# Case 3: three clusters, different sizes
{"partition":[0,1,0,2,1]}
@ cluster 0: [0,2]; cluster 1: [1,4]; cluster 2: [3]
{"groups":{"0":[0,2],"1":[1,4],"2":[3]}}

# Case 4: non-zero/unsorted labels
{"partition":[2,2,5,2]}
@ cluster 2: [0,1,3]; cluster 5: [2]
{"groups":{"2":[0,1,3],"5":[2]}}

# Case 5: single element
{"partition":[7]}
@ one node 0 in cluster 7
{"groups":{"7":[0]}}
