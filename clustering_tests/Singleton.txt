# Case 1: single element, always singleton
{"node":0,"partition":[0]}
@ only one node in the partition → it is a singleton
{"singleton":true}

# Case 2: node in a cluster of size 2
{"node":0,"partition":[1,1]}
@ both nodes 0 and 1 have cluster ID 1 → node 0 is not singleton
{"singleton":false}

# Case 3: node in a cluster of size 1, others in a different cluster
{"node":0,"partition":[0,1,1]}
@ cluster 0: {0}, cluster 1: {1,2} → node 0 is singleton
{"singleton":true}

# Case 4: last node is singleton
{"node":2,"partition":[5,5,7]}
@ cluster 5: {0,1}, cluster 7: {2} → node 2 is singleton
{"singleton":true}

# Case 5: all nodes in same cluster
{"node":1,"partition":[3,3,3]}
@ cluster 3 has three members → node 1 is not singleton
{"singleton":false}
